## Usage

```tsx
import { Search } from "lucide-react";

import {
  InputGroup,
  InputGroupAddon,
  InputGroupInput,
} from "@/components/ui/input-group";
```

```tsx
<InputGroup>
  <InputGroupAddon>
    <Search aria-hidden="true" />
  </InputGroupAddon>
  <InputGroupInput aria-label="Search components" placeholder="Search components..." />
</InputGroup>
```
