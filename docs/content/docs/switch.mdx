## Usage

```tsx
"use client";
import { useState } from "react";
import { Switch } from "@/components/ui/switch";

export default function NotificationSetting() {
  const [enabled, setEnabled] = useState(false);
  return (
    <div className="grid gap-3">
      <div className="flex items-center gap-3">
        <Switch id="notifications" checked={enabled} onCheckedChange={setEnabled} />
        <label htmlFor="notifications">Email notifications</label>
      </div>
      <p role="status">Preview setting: {enabled ? "on" : "off"}.</p>
    </div>
  );
}
```

## API reference

| Prop | Purpose | Default |
| --- | --- | --- |
| `checked` / `onCheckedChange` | Controlled boolean state | Uncontrolled |
| `defaultChecked` | Initial state | `false` |
| `size` | default or sm | `default` |
| `name` / `value` | Form key and checked value | Native behavior |
| `disabled` / `required` | Interaction and form constraints | `false` |
| `id` | Visible label target | Generated if omitted |

## Accessibility

Use a stable label describing the setting, not changing action text such as Enable/Disable. Do not rely only on color to show state. Show pending or error state for server updates and never imply an unsaved change has been persisted.
