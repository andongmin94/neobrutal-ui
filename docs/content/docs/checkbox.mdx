## Usage

```tsx
"use client";
import { useState } from "react";
import { Checkbox } from "@/components/ui/checkbox";

export default function ConsentField() {
  const [accepted, setAccepted] = useState(false);
  return (
    <div className="grid gap-3">
      <div className="flex items-center gap-2">
        <Checkbox
          id="terms"
          checked={accepted}
          onCheckedChange={(value) => setAccepted(value === true)}
        />
        <label htmlFor="terms">Accept the terms</label>
      </div>
      <p role="status">{accepted ? "Accepted in this preview." : "Not accepted."}</p>
    </div>
  );
}
```

## API reference

| Prop / part | Purpose | Default |
| --- | --- | --- |
| `checked` / `defaultChecked` | Boolean or `indeterminate` state | Unchecked |
| `onCheckedChange` | New checked state and event details | — |
| `indeterminate` | Explicit mixed state | Derived from checked value |
| `disabled` / `required` | Interaction and form constraints | `false` |
| `name` / `value` / `form` | Form association through a hidden input | — |
| `id` / `inputRef` | Label association and hidden input reference | — |

## Accessibility

Associate a visible label with the checkbox. Space toggles the focused control. Use mixed state for partial selection, not as a third agreement option. Checked state does not submit or save anything on its own. When mapping checked values to a boolean, compare with `true` rather than coercing the string `indeterminate` to true.
